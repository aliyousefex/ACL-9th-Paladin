//1. Query 1
MATCH (d:Driver)<-[:BY_DRIVER]-(res:Result)-[:FOR_RACE]->(r:Race)-[:IN_SEASON]->(:Season {year: 2021})
WITH d, sum(res.points) AS Points, count(CASE WHEN res.position = 1 THEN 1 END) AS Wins
RETURN d.forename + ' ' + d.surname AS Driver, Points, Wins
ORDER BY Points DESC, Wins DESC, Driver ASC

//2. Query 2
MATCH (c:Constructor)<-[:FOR_CONSTRUCTOR]-(res:Result)-[:FOR_RACE]->(r:Race)-[:IN_SEASON]->(s:Season)
WHERE s.year >= 2014 AND s.year <= 2021
MATCH (res)-[:HAS_STATUS]->(st:Status)
WITH c, count(res) AS Entries, count(CASE WHEN st.status = 'Finished' OR st.status STARTS WITH '+' THEN 1 END) AS Classified
WHERE Entries >= 100
RETURN c.constructor_name AS Constructor, Entries, Classified, toFloat(Classified) / Entries AS Reliability
ORDER BY Reliability DESC, Constructor ASC

//3. Query 3
MATCH (h:Result)-[:BY_DRIVER]->(:Driver {driverId: 1}),
      (b:Result)-[:BY_DRIVER]->(:Driver {driverId: 822}),
      (h)-[:FOR_RACE]->(r:Race)<-[:FOR_RACE]-(b),
      (h)-[:FOR_CONSTRUCTOR]->(c:Constructor)<-[:FOR_CONSTRUCTOR]-(b)
RETURN c.constructor_name AS Constructor,
       count(*) AS Shared_Races,
       count(CASE WHEN h.positionOrder < b.positionOrder THEN 1 END) AS Hamilton_Ahead,
       count(CASE WHEN b.positionOrder < h.positionOrder THEN 1 END) AS Bottas_Ahead
